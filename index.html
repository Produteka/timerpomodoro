// Timer durations in seconds


const durations = {


work: 25 * 60,


shortBreak: 5 * 60,


longBreak: 15 * 60


};




// Update timer display


function updateDisplay() {


const minutes = Math.floor(timeLeft / 60);


const seconds = timeLeft % 60;


timeDisplay.textContent = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;




// Update progress ring


const totalTime = durations[mode];


const dashoffset = 628.3 * (1 - timeLeft / totalTime);


progressIndicator.style.strokeDashoffset = dashoffset;


}




// Get random phrase


function getRandomPhrase() {


const randomIndex = Math.floor(Math.random() * inspirationalPhrases.length);


return inspirationalPhrases[randomIndex];


}




// Display a new phrase with animation


function showNewPhrase() {


phraseDisplay.style.opacity = 0;


setTimeout(() => {


const phrase = getRandomPhrase();


const phraseText = phraseDisplay.querySelector('.phrase-text');


const phraseAuthor = phraseDisplay.querySelector('.phrase-author');




phraseText.textContent = phrase.text;


phraseAuthor.textContent = `— ${phrase.author}`;




phraseDisplay.classList.add('fade-in');


phraseDisplay.style.opacity = 1;


setTimeout(() => {


phraseDisplay.classList.remove('fade-in');


}, 500);


}, 300);


}

// Start timer


function startTimer() {


if (!isRunning) {


isRunning = true;


startBtn.classList.add('hidden');


pauseBtn.classList.remove('hidden');




timer = setInterval(() => {


timeLeft--;


updateDisplay();




if (timeLeft <= 0) {


clearInterval(timer);


isRunning = false;




// Play notification sound


playTimerEndSound();




if (mode === 'work') {


sessionsCompleted++;


sessionCount.textContent = sessionsCompleted;




if (sessionsCompleted % 4 === 0) {


setMode('longBreak');


} else {


setMode('shortBreak');


}


} else {


setMode('work');


}




showNewPhrase();


startBtn.classList.remove('hidden');


pauseBtn.classList.add('hidden');


}


}, 1000);


}


}




// Pause timer


function pauseTimer() {


if (isRunning) {


clearInterval(timer);


isRunning = false;


startBtn.classList.remove('hidden');


pauseBtn.classList.add('hidden');


}


}




// Reset timer


function resetTimer() {


clearInterval(timer);


isRunning = false;


timeLeft = durations[mode];


updateDisplay();


startBtn.classList.remove('hidden');


pauseBtn.classList.add('hidden');


}




// Set timer mode


function setMode(newMode) {


mode = newMode;


timeLeft = durations[mode];


updateDisplay();




// Update button styles


[workBtn, shortBreakBtn, longBreakBtn].forEach(btn => {


btn.classList.remove('active');


});




// Set active button style and timer circle color


const timerCircle = document.querySelector('.timer-circle');




if (mode === 'work') {


workBtn.classList.add('active');


workBtn.style.backgroundColor = '#FF7700';


timerCircle.style.background = 'linear-gradient(135deg, #FF7700, #FA55DE)';


} else if (mode === 'shortBreak') {


shortBreakBtn.classList.add('active');


shortBreakBtn.style.backgroundColor = '#FA55DE';


timerCircle.style.background = 'linear-gradient(135deg, #FA55DE, #7e06c8)';


} else {


longBreakBtn.classList.add('active');


longBreakBtn.style.backgroundColor = '#7e06c8';


longBreakBtn.style.color = 'white';


timerCircle.style.background = 'linear-gradient(135deg, #7e06c8, #FF7700)';


}




// Reset timer if it was running


if (isRunning) {


clearInterval(timer);


isRunning = false;


startBtn.classList.remove('hidden');


pauseBtn.classList.add('hidden');


}


}




// Toggle dark mode


function toggleDarkMode() {


isDarkMode = darkModeToggle.checked;


if (isDarkMode) {


document.body.classList.add('dark-mode');


document.querySelector('.logo-container').classList.add('dark-mode');


} else {


document.body.classList.remove('dark-mode');


document.querySelector('.logo-container').classList.remove('dark-mode');


}


}




// Event listeners

workBtn.addEventListener('click', () => setMode('work'));


shortBreakBtn.addEventListener('click', () => setMode('shortBreak'));


longBreakBtn.addEventListener('click', () => setMode('longBreak'));


darkModeToggle.addEventListener('change', toggleDarkMode);




// Initialize


updateDisplay();
